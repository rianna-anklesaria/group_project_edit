central_area_data <- central_area_data %>% filter(area < 4000)

central_data_with_eras <- central_area_data %>%
  mutate(
    era_built = case_when(
      Year.Built < 1913 ~ "Victorian and Gilded age",
      Year.Built > 1913 & Year.Built <= 1945 ~ "Interwar",
      Year.Built > 1945 & Year.Built < 1980 ~ "Post-war",
      Year.Built >= 1980 ~ "20th century or modern"
    ), 
    era_remodelled = case_when(
      Year.Remod.Add < 1960 ~ "1950s",
      Year.Remod.Add >= 1960 & Year.Remod.Add < 1980 ~ "1960s and 1970s",
      Year.Remod.Add >= 1980 & Year.Remod.Add < 2000 ~ "1980s and 1990s",
      Year.Remod.Add >= 2000 ~ "2000s"
    )
  )

view(central_data_with_eras)

central_data_with_eras <- central_data_with_eras %>%
  mutate(price_k = price/1000)

central_data_with_eras %>%
  ggplot(aes(x = era_built, y = price_k)) +
  geom_boxplot() +
  labs(title = "Price of Houses by Era Built",
       x = "Era Built",
       y = "Price in Thousands $") +
  theme_minimal()

central_data_with_eras %>%
  ggplot(aes(x = era_remodelled, y = price_k)) +
  geom_boxplot() +
  labs(title = "Price of Houses by Era Remodelled",
       x = "Era Remodelled",
       y = "Price in Thousands $") +
  theme_minimal()

central_data_with_eras %>%
  ggplot(aes(x = area, y = price_k)) +
  geom_point(aes(colour = House.Style)) +
  geom_smooth(method = "lm", se = FALSE, colour = "black") +
  labs(title = "House Prices by Area",
       subtitle = "by Style",
       x = "Area in square feet",
       y = "Price in Thousands $") +
  theme_minimal()
